%% 本書的語言：拉丁文為主體，希臘文以 \grc 切換 (由 md2tex.py 自動加上)

%*====================================================================
%*  詞條格式
%*====================================================================
% 疏排 (Sperrung)：原書以字距加寬標示詞條
\newcommand{\lemma}[1]{{\addfontfeatures{LetterSpace=12}#1}}

% Bekker 頁碼的 a/b 欄
% 用字型本身的 superior 字形 (Alegreya 有 sups)
\newcommand{\bekkercol}[1]{{\addfontfeatures{VerticalPosition=Superior}#1}}
\newcommand{\ba}{\bekkercol{a}}
\newcommand{\bb}{\bekkercol{b}}

% 書眉：每頁第一個與最後一個詞條
\NewMarkClass{lemma}
\newcommand{\lemmamark}[1]{\InsertMark{lemma}{#1}}

\newlength{\entryindent} \setlength{\entryindent}{1em}
% 詞條：懸掛縮排
\newcommand{\entry}[1]{%
  \par\noindent\hangindent=\entryindent\hangafter=1
  \if\relax\detokenize{#1}\relax\else\lemmamark{#1}\fi}
% 詞條內分項：首行縮排
\newcommand{\subpar}{\par\indent}

% 字母分節
\newcommand{\lettersection}[1]{%
  \end{multicols}
  \begin{center}\Large\grc{#1}\end{center}\vspace{0.3\baselineskip}
  \begin{multicols}{2}}

% 原書頁碼
\newcommand{\origpage}[1]{%
  \ifshoworigpage\hbox{}\nobreak
    \textcolor{gray}{\textsf{\scriptsize[#1]}}\fi}
% 不確定讀法
\newcommand{\unsure}[1]{%
  \ifshowunsure\colorbox{yellow!40}{#1}\else#1\fi}

%*====================================================================
%*  書眉
%*====================================================================
\pagestyle{fancy}

\renewcommand{\headrulewidth}{0pt}

  \hspace{\columnsep}%
  \makebox[\columnwidth][c]{\grc{\LastMark{lemma}}}}
\fancyhead[LE,RO]{\thepage}
